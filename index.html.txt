    <script>
        const tg = window.Telegram.WebApp;
        tg.expand(); // Раскрываем мини-аппу на весь экран

        function switchTab(tab) {
            const workoutsTab = document.getElementById('tab-workouts');
            const historyTab = document.getElementById('tab-history');
            const workoutsContent = document.getElementById('content-workouts');
            const historyContent = document.getElementById('content-history');

            if (tab === 'workouts') {
                workoutsTab.className = "flex-1 py-2 text-sm font-semibold rounded-xl bg-indigo-600 text-white transition";
                historyTab.className = "flex-1 py-2 text-sm font-semibold rounded-xl text-slate-400 hover:text-white transition";
                workoutsContent.classList.remove('hidden');
                historyContent.classList.add('hidden');
            } else {
                historyTab.className = "flex-1 py-2 text-sm font-semibold rounded-xl bg-indigo-600 text-white transition";
                workoutsTab.className = "flex-1 py-2 text-sm font-semibold rounded-xl text-slate-400 hover:text-white transition";
                historyContent.classList.remove('hidden');
                workoutsContent.classList.add('hidden');
                renderHistory();
            }
        }

        function completeWorkout(name) {
            const now = new Date();
            const dateStr = now.toLocaleDateString('ru-RU', { day: 'numeric', month: 'short' }) + ' в ' + now.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
            
            let history = JSON.parse(localStorage.getItem('workout_history') || '[]');
            history.unshift({ name, date: dateStr });
            localStorage.setItem('workout_history', JSON.stringify(history));

            if (tg && tg.HapticFeedback) {
                tg.HapticFeedback.notificationOccurred('success');
            }

            alert(`🔥 Отлично! Тренировка "${name}" успешно записана!`);
            switchTab('history');
        }

        function renderHistory() {
            const listEl = document.getElementById('history-list');
            let history = JSON.parse(localStorage.getItem('workout_history') || '[]');
            
            if (history.length === 0) {
                listEl.innerHTML = `<div class="text-center py-12 text-slate-500 text-sm">Пока нет завершенных тренировок.<br>Самое время начать! 🚀</div>`;
                return;
            }

            listEl.innerHTML = history.map(item => `
                bg-slate-900 border border-slate-800 rounded-xl p-3 flex justify-between items-center text-sm
                <div class="bg-slate-900 border border-slate-800 rounded-xl p-3 flex justify-between items-center text-sm w-full">
                    <span class="font-medium text-white">${item.name}</span>
                    <span class="text-xs text-slate-400">${item.date}</span>
                </div>
            `).join('');
        }

        function clearHistory() {
            if (confirm('Очистить всю историю?')) {
                localStorage.removeItem('workout_history');
                renderHistory();
            }
        }
    </script>
